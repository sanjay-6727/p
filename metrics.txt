 B1. FULL FLOWMONITOR BLOCK: throughput, delay, jitter, loss, PDR,
     lost packets, fairness. Paste after Simulator::Run() and

    Ptr<Ipv4FlowClassifier> cl = DynamicCast<Ipv4FlowClassifier>(fmh.GetClassifier());
    double sumThr = 0, sumSq = 0; int nFlows = 0;

    for (auto &kv : mon->GetFlowStats()) {
        Ipv4FlowClassifier::FiveTuple t = cl->FindFlow(kv.first);
        const auto &s = kv.second;

        // keep only the data flows (skip the TCP ACK flows going back)
        if (t.destinationPort != 8080 && t.destinationPort != 5000 &&
            t.destinationPort != 9000) continue;                 // your ports

        double dur  = (s.timeLastRxPacket - s.timeFirstTxPacket).GetSeconds();
        double thr  = dur > 0 ? s.rxBytes * 8.0 / dur / 1e6 : 0;           // Mbps
        double dly  = s.rxPackets > 0 ? s.delaySum.GetSeconds() / s.rxPackets * 1000 : 0;   // ms
        double jit  = s.rxPackets > 1 ? s.jitterSum.GetSeconds() / (s.rxPackets - 1) * 1000 : 0;  // ms
        double loss = s.txPackets > 0 ? 100.0 * (s.txPackets - s.rxPackets) / s.txPackets : 0;    // %
        double pdr  = 100.0 - loss;                                         // %

        std::cout << "Flow " << kv.first << " (" << t.sourceAddress << " -> "
                  << t.destinationAddress << ":" << t.destinationPort << ")\n"
                  << "  Tx packets   : " << s.txPackets << "\n"
                  << "  Rx packets   : " << s.rxPackets << "\n"
                  << "  Lost packets : " << s.lostPackets << "\n"
                  << "  Throughput   : " << thr << " Mbps\n"
                  << "  Avg delay    : " << dly << " ms\n"
                  << "  Jitter       : " << jit << " ms\n"
                  << "  Packet loss  : " << loss << " %\n"
                  << "  PDR          : " << pdr << " %\n";

        sumThr += thr; sumSq += thr * thr; nFlows++;
    }
    // Jain's fairness index (1.0 = perfectly fair)
    if (nFlows) std::cout << "Fairness = " << sumThr * sumThr / (nFlows * sumSq) << "\n";

    double thr = 0, dly = 0, tx = 0, rx = 0, n = 0;

---------------------------------------------------------------------
 B8. FORMULAS (write these in the report)
---------------------------------------------------------------------
 Throughput (Mbps) = rxBytes x 8 / (timeLastRx - timeFirstTx) / 10^6
 Avg delay (ms)    = delaySum / rxPackets x 1000
 Jitter (ms)       = jitterSum / (rxPackets - 1) x 1000
 Packet loss (%)   = (txPackets - rxPackets) / txPackets x 100
 PDR (%)           = rxPackets / txPackets x 100
 Fairness (Jain)   = (sum xi)^2 / (n x sum xi^2)
 UDP interval (s)  = packetSizeBytes x 8 / dataRateBps
 RTT (C sockets)   = time(reply received) - time(request sent)
